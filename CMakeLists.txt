cmake_minimum_required(VERSION 3.20)
project(RA2311003020323_Workbench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(workbench src/main.cpp src/engine.cpp)
target_link_libraries(workbench PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(workbench PRIVATE /W4 /permissive-)
else()
  target_compile_options(workbench PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_executable(workbench_tests tests/tests.cpp src/engine.cpp)
target_include_directories(workbench_tests PRIVATE src)
target_link_libraries(workbench_tests PRIVATE Threads::Threads)
add_test(NAME workbench_tests COMMAND workbench_tests)
